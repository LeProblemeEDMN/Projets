#version 330

in vec2 tCoords;//texture coords
in vec3 surfaceNormal;//normal
in vec3 toCameraVector;//vector from camera to the object
in float visibility;//visibility of the object (to mix with fog)
in vec3 toPointVector[5];
in vec3 worldPos;//position of the fragment
in vec4[4] shadowCoords;//shadow coords of the
in float id_shadow;//distance to the camera

out vec4 out_Color;
//spot light structure
struct Spot {
	vec3 attenuation;
	vec3 lookVec;
	float angle;
	vec3 color;
	sampler2D textSpot;
	float far;
};

//point light structure
struct PointLight {
	vec3 attenuation;
	vec3 color;
	float far;
};

//parameter of the cascaded shadow map used to find the id of the shadow map used to draw shadows
uniform float d0_shadow;
uniform float r_shadow;

//lights
uniform PointLight[10] points;
uniform Spot[10] spots;

//positions and shadow matrix of the spots
uniform vec3[10] positionSpot;
uniform mat4[10] spotMatrix;

//light used
uniform float[6] spotInUse;
uniform float[6] pointInUse;

//points light shadow textures
uniform samplerCube[10] textCubes;

//texture of the model
uniform sampler2D modelTexture;
//ambient occlusion/roughness/metal(ARM) of the model
uniform sampler2D ARMtexture;
//normal map of the model
uniform sampler2D normalMap;
//ambient occlusion texture
uniform sampler2D ssaoTexture;
//cascaded shadow map textures
uniform sampler2D[4] sunTexture;
//parameters of the Blinn phong model
uniform float reflectivity_max;
uniform float shineDamper_max;

//color of the fog.
uniform vec3 skyColor;
//screen size
uniform vec2 screenSize;
//shadow map size
uniform vec2 mapSize;


//parameters of PCF (soft shadow algorithm).
const int pcfCount=3;
const float totalTexels=(pcfCount*2.0+1.0)*(pcfCount*2.0+1.0);

float sunShadow(int id);

const float near=0.1;
float LinearizeDepth(float depth,float far)
{
    float z = depth * 2.0 - 1.0; // back to NDC

    return (2.0 * near * far) / (far + near - z * (far - near));
}

//TODO
//pr les spots transformer les pos sur la shadow map dans le vertex shader et pas dans le fragment shader
void main(void){
    //check que la texture n'est pas transparente
	vec4 textureColour=texture(modelTexture,tCoords );
	if(textureColour.a<0.5 ){
		discard;
	}

    vec3 totalDiffuse= vec3(0.0);
    vec3 totalSpecular= vec3(0.0);

    //calcule quelle shadow map doit être utilise à partir de la distance à al camera
    int id_shadow_int=int(log(1-id_shadow/d0_shadow*(1-r_shadow))/log(r_shadow));

    vec3 unitNormal=surfaceNormal;
    vec3 unitCameraVector=normalize(toCameraVector);

    //sample the AO map:
    vec4 arm_values=texture(ARMtexture,tCoords );
    float shineDamper=1/(arm_values.y*arm_values.y+0.0000001);
    float reflectivity=reflectivity_max*arm_values.z+(1-arm_values.z)*(1-arm_values.y);
	//-----------SPOTS------------------------------------------------
	//On peut utiliser au maximum 5 spots light/objet
    for(int id=0;id<int(spotInUse[5]);id++){
        int i=int(spotInUse[id]);

        vec3 dirVec=spots[i].lookVec;
        vec3 toLightVector=positionSpot[i] -worldPos;
        vec3 unitLightVector=normalize(toLightVector);
        float ac=acos(dot(-unitLightVector,dirVec));
        //on check qu'on est bien dans l'angle de vue du spot.
        if(ac< spots[i].angle){
        float shadowFactor=1;

        //on calcule l'ombre du spot
        vec4 s=spotMatrix[i]*vec4(worldPos,1);
        s.xyz/=s.w;
        s.xyz*=0.5;
        s.xyz+=0.5;

        float tz=texture(spots[i].textSpot,s.xy).r;
        float z=LinearizeDepth(s.z,spots[i].far)/spots[i].far;
        //si l'objet est dans l'ombre
        if((z-tz)>0.005){
           shadowFactor=0.7;
        }
        //BLinn-phong lightning
        float distance=length(toLightVector);
        float attenuationFactor=spots[i].attenuation.x+(spots[i].attenuation.y*distance)+(spots[i].attenuation.z*distance*distance);

        float nDot1=dot(unitNormal,unitLightVector);
        float brightness=max(nDot1,0.0);
        vec3 lightDirection=-unitLightVector;
        vec3 reflectedLightDirection=reflect(lightDirection,unitNormal);

        float specularFactor=dot(reflectedLightDirection,unitCameraVector);
        specularFactor=max(specularFactor,0.0);
        float dampedFactor=pow(specularFactor,shineDamper);
        //accumulate the light
        totalDiffuse=totalDiffuse+(brightness*spots[i].color)/(attenuationFactor/shadowFactor);
        totalSpecular=totalSpecular+(dampedFactor*reflectivity*spots[i].color)/(attenuationFactor/shadowFactor);
        }
    }

	//-----------POINTS --------------------------------------
	// on peut utiliser jusqu'à 5 point light. (0 est le soleil).
	for(int id=0;id<int(pointInUse[5]);id++){
        int i=int(pointInUse[id]);

            vec3 unitLightVector=normalize(toPointVector[i]);

            float shadowFactor=1;
            float distance=length(toPointVector[i]);
            float z=(distance-0.1)/points[i].far;
            float tz=texture(textCubes[i],normalize(-toPointVector[i])).r;
            float v=z-tz;

            //on test l'ombre d'un point normal
            if(v>0.005){
                shadowFactor=0.3;
            }
            if(i==0)shadowFactor=sunShadow(id_shadow_int);
            //facteur d'attenuation de la lumière.
            float attenuationFactor=points[i].attenuation.x+(points[i].attenuation.y*distance)+(points[i].attenuation.z*distance*distance);

            //Blinn-phong lightning
            float nDot1=dot(unitNormal,unitLightVector);
            float brightness=max(nDot1,0.0);
            vec3 lightDirection=-unitLightVector;
            vec3 reflectedLightDirection=reflect(lightDirection,unitNormal);

            float specularFactor=dot(unitNormal,normalize(unitLightVector+unitCameraVector));//dot(reflectedLightDirection,unitCameraVector);
            specularFactor=max(specularFactor,0.0);
            float dampedFactor=pow(specularFactor,shineDamper);
            //accumulate the light
            totalDiffuse=totalDiffuse+(brightness*points[i].color)/(attenuationFactor/shadowFactor);
            totalSpecular=totalSpecular+(dampedFactor*reflectivity*points[i].color)/(attenuationFactor/shadowFactor);
	}
	//---------------------FIN LUMIERES-----------------------------------

    //ambient occlusion obtained with SSAO
    float ambientOcclusion=texture(ssaoTexture,gl_FragCoord.xy/screenSize ).r*pow(arm_values.x,2);

	out_Color = textureColour*0.3*ambientOcclusion+0.7*vec4(totalDiffuse,0.0)*textureColour+vec4(totalSpecular,1.0);
    //mix the light color and the sky color
	out_Color =mix(vec4(skyColor,1.0),out_Color,visibility);

}

//Compute the shadow with shadow mapping
float sunShadow(int id_map){
//check if we are inside the box.
	if(shadowCoords[id_map].x<0 || shadowCoords[id_map].x>1  ||shadowCoords[id_map].y<0 || shadowCoords[id_map].y>1 || shadowCoords[id_map].z<0 )return 1;//|| shadowCoords[id_map].z>1
	//if it is not the closest shadow map we dont do soft shadows
	if(id_map>0){
	    float objectNearestLight=texture(sunTexture[id_map],shadowCoords[id_map].xy).r;
        if(shadowCoords[id_map].z-0.005<objectNearestLight){
        	return 1;
        }
        return 0;
	}
	//soft shadows algorithm
	float texelSizeX=1.0/mapSize.x;
	float texelSizeY=1.0/mapSize.y;
	float total=0.0;
	for(int x=-pcfCount;x<=pcfCount;x++){
		for(int y=-pcfCount;y<=pcfCount;y++){
			float objectNearestLight=texture(sunTexture[id_map],shadowCoords[id_map].xy+vec2(x*texelSizeX,y*texelSizeY)).r;
			if(shadowCoords[id_map].z-0.001<objectNearestLight){
				total+=1;
			}
	    }
	}
	total/=totalTexels;
	return total;
}
