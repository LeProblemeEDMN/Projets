#version 330 core

in vec3 position;
in vec2 textureCoords;
in vec3 normal;
layout (location=5) in mat4 transformationMatrix;

out vec2 tCoords;//texture coords
out vec3 surfaceNormal;//normal
out vec3 toCameraVector;//vector from camera to the object
out float visibility;//visibility of the object (to mix with fog)
out vec3 toPointVector[5];
out vec3 worldPos;//position of the fragment
out vec4[4] shadowCoords;//shadow coords of the
out float id_shadow;//distance to the camera

//proj and view matrix
uniform mat4 projectionMatrix;
uniform mat4 viewMatrix;
uniform mat4 invViewMatrix;
//shadow space matrix
uniform mat4 toShadowMapSpace[4];

//position of point lights
uniform vec3[10] positionPoint;
//points lights used
uniform float[6] pointInUseVertex;
//does the mode luse fake lightning (like grass for example).
uniform float useFakeLightning;

//fog parameters
uniform float density;
uniform float gradient;

void main(void){
	tCoords=textureCoords;
    //compute the screen coords
	vec4 modelPos=vec4(position,1.0);
	vec4 worldPosition=transformationMatrix*modelPos;
	worldPos=worldPosition.xyz;
	vec4 positionRelativToCamera=viewMatrix*worldPosition;
	gl_Position =projectionMatrix* positionRelativToCamera;

    // if we use fake lightning the normal is (0,1,0).
	vec3 actualNormal=(1-useFakeLightning)*normal+vec3(0.0,useFakeLightning,0.0);

	surfaceNormal=(transformationMatrix*vec4(actualNormal,0.0)).xyz;
	surfaceNormal=normalize(surfaceNormal);

	toCameraVector=(invViewMatrix*vec4(0.0,0.0,0.0,1.0)).xyz - worldPosition.xyz;

    //compute the visibility of the object.
	float distance=length(positionRelativToCamera.xyz);
	visibility=exp(-pow((distance*density),gradient));
	visibility=clamp(visibility,0.0,1.0);

    //compute the point positions
	for(int i=0;i<int(pointInUseVertex[5]);i++){
		int id=int(pointInUseVertex[i]);
		//if(i>=0){
		toPointVector[i]=positionPoint[id] -worldPosition.xyz;
		//}
	}

    //cascaded shadow map
    id_shadow=-positionRelativToCamera.z;
    for(int i=0;i<4;i++)shadowCoords[i]=toShadowMapSpace[i]*worldPosition;


}
